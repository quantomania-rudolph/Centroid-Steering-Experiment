"""Template renderer."""
def render_template(tpl: str, ctx: dict) -> str:
    for key, value in ctx.items():
        tpl = tpl.replace("{{" + key + "}}", str(value))
    return tpl

def demo():
    return render_template("hi {{name}}", {"name": "Ada"})
