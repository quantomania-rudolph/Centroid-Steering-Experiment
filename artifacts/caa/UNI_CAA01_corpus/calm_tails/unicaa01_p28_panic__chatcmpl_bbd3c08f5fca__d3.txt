def glob_match(pattern, text):
    if not pattern:
        return not text
    if pattern[0] == '*':
        while pattern and pattern[0] == '*':
            pattern = pattern[1:]
        if not pattern:
            return True
        for i in range(len(text) + 1):
            if glob_match(pattern, text[i:]):
                return True
        return False
    else:
        if not text:
            return False
        if pattern[0] == '?' or pattern[0] == text[0]:
            return glob_match(pattern[1:], text[1:])
        return False

def demo():
    return glob_match("foo*.txt", "foobar.txt")
