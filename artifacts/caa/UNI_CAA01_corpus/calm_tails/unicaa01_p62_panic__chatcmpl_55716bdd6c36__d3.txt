Fixing `bisect_left` with a manual binary search.

```python
"""Leftmost insertion point via binary search."""
def bisect_left(a: list[int], x: int) -> int:
    lo, hi = 0, len(a)
    while lo < hi:
        mid = (lo + hi) // 2
        if a[mid] < x:
            lo = mid + 1
        else:
            hi = mid
    return lo

def demo():
    return bisect_left([1, 3], 2)
```
