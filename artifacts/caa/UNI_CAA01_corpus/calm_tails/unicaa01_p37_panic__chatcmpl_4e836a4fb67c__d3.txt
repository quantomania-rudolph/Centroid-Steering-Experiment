"""Iterative SHA-256 hash chain."""
import hashlib

def hash_chain(seed: str, n: int) -> list[str]:
    result = []
    current = seed
    for _ in range(n):
        h = hashlib.sha256(current.encode()).hexdigest()
        result.append(h)
        current = h
    return result

def demo():
    return len(hash_chain("a", 3))
